```python
def binary_search(nums, target):
    left = 0
    right = len(nums) - 1

    while left <= right:
        mid = (left + right) // 2

        if nums[mid] == target:
            return mid
        elif nums[mid] < target:
            left = mid + 1   # 中间值太小，排除左半边和中点
        else:
            right = mid - 1  # 中间值太大，排除右半边和中点

    return -1
```
